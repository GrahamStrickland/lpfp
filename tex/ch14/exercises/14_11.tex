Applying (14.16) repeatedly to the given figures, we have
\begin{equation*}
    \begin{split}
        v(0.1\,\text{s}) &= v(0\,\text{s}) + \frac{F^{tv}_{net}\bigl(0\,\text{s}, v(0\,s)\bigr) 
            (0.1\,\text{s})}{1 \text{kg}} \\
                         &= 2.0000 \, \text{m}/\text{s} + \bigl(F_1(0) 
                            + F_2(2.0000)\text{m}/\text{s}^2 \bigr) (0.1\,\text{s}) \\
                         &= 2.0000 \, \text{m}/\text{s} 
                            + \bigl(4\cos{0} - 3(2.0000)\text{m}/\text{s}^2 \bigr) (0.1\, \text{s}) \\
                         &= 1.8000 \, \text{m}/\text{s} \\
        v(0.2\,\text{s}) &= v(0.1\,\text{s}) + \frac{F^{tv}_{net}\bigl(0.1\,\text{s}, v(0.1\,s)\bigr) 
            (0.1\,\text{s})}{1 \text{kg}} \\
                         &= 1.8000 \, \text{m}/\text{s} + \bigl(F_1(0.1) 
                            + F_2(1.8000)\text{m}/\text{s}^2 \bigr) (0.1\,\text{s}) \\
                         &= 1.8000 \, \text{m}/\text{s} 
                            + \bigl(4\cos{0.1} - 3(1.8000)\text{m}/\text{s}^2 \bigr) (0.1\, \text{s}) \\
                         &= 1.6580 \, \text{m}/\text{s} \\
        v(0.3\,\text{s}) &= v(0.2\,\text{s}) + \frac{F^{tv}_{net}\bigl(0.2\,\text{s}, v(0.2\,s)\bigr) 
            (0.1\,\text{s})}{1 \text{kg}} \\
                         &= 1.6580 \, \text{m}/\text{s} + \bigl(F_1(0.2) 
                            + F_2(1.6580)\text{m}/\text{s}^2 \bigr) (0.1\,\text{s}) \\
                         &= 1.6580 \, \text{m}/\text{s} 
                            + \bigl(4\cos{0.2} - 3(1.6580)\text{m}/\text{s}^2 \bigr) (0.1\, \text{s}) \\
                         &= 1.5526 \, \text{m}/\text{s}.
    \end{split}
\end{equation*}
