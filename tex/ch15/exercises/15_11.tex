We have the following results using the Euler-Cromer method with
\begin{equation*}
    \begin{split}
        F_{\text{net}}^{txv}(t, x_0, v_0) &= -kx_0 - \frac{1}{2}C\rho A \lvert v_0 \rvert v_0 - mg \\
        &= (-0.8\,\text{kg}/\text{s}^2)(x_0\,\text{m}) 
            - \frac{1}{2} \cdot 2 \cdot (1.225\,\text{kg}/\text{m}^3) \\
        &\qquad (\pi \cdot {0.02}^2\,\text{m}^2)\lvert v_0 \rvert (v_0\,\text{m}/\text{s}^2) 
            - (0.0027\,\text{kg})(9.80665\,\text{m}/\text{s}^2)
    \end{split}
\end{equation*}
for initial values $t = 0.0\,\text{s}$, $x(0.0\,\text{s}) = 0.1000\,\text{m}$ and 
$v(0.0\,\text{s}) = 0.0000\,\text{m/s}$.

\begin{equation*}
    \begin{split}
        v(0.1\,\text{s}) &= v(0\,\text{s}) 
            + \frac{F_{\text{net}}^{txv}(0\,\text{s}, x(0\,\text{s}), v(0\,\text{s}))}{m}\Delta t \\
                         &= 0.0000\,\text{m/s} 
            + \frac{(-0.106478\,\text{N})(0.1\,\text{s})}{0.0027\,\text{kg}} \\
                         &= -3.9436\,\text{m/s} \\
        x(0.1\,\text{s}) &= x(0\,\text{s}) + v(0.1\,\text{s}) \Delta t \\
                         &= 0.1000\,\text{m} + (-3.9436\,\text{m/s})(0.1\,\text{s}) \\
                         &= -0.2944\,\text{m} \\
        v(0.2\,\text{s}) &= v(0.1\,\text{s}) 
            + \frac{F_{\text{net}}^{txv}(0.1\,\text{s}, x(0.1\,\text{s}), v(0.1\,\text{s}))}{m}\Delta t \\
                         &= -3.9436\,\text{m/s} 
            + \frac{(-0.082538\,\text{N})(0.1\,\text{s})}{0.0027\,\text{kg}} \\
                         &= -3.9436\,\text{m/s} \\
        x(0.2\,\text{s}) &= x(0\,\text{s}) + v(0.2\,\text{s}) \Delta t \\
                         &= 0.1000\,\text{m} + (-3.9436\,\text{m/s})(0.1\,\text{s}) \\
                         &= -0.2944\,\text{m} \\
    \end{split}
\end{equation*}
